function ChangeSelectedPanelDuration()
{
  var project = new StoryboardManager;
  var selection = new SelectionManager;
  var CURRENTPANEL = selection.getPanelSelection();
/* var PANELDIALOGUE = new CaptionManager;  */

  if ( CURRENTPANEL.length <= 0 ) return;

/*  QUERRY POPUP How Many Extra Milliseconds to Add at Tail of Each Voice Clip?*/
  QUERRYPANELMILISECONDSTOTALTAILOFFSET = new Dialog;
   QUERRYPANELMILISECONDSTOTALTAILOFFSET.title = " Set  Extra Milliseconds to Add at Tail of Each Voice Clip?";
  var PANELMILISECONDSTOTALTAILOFFSET = new NumberEdit;
  PANELMILISECONDSTOTALTAILOFFSET.text = "How Many Extra Milliseconds to Add at Tail of Each Voice Clip?";
  PANELMILISECONDSTOTALTAILOFFSET.minimum = 0;
  PANELMILISECONDSTOTALTAILOFFSET.maximum =  10000 ;
  PANELMILISECONDSTOTALTAILOFFSET.value = project.getPanelDuration ( CURRENTPANEL[0] );
 QUERRYPANELMILISECONDSTOTALTAILOFFSET.add( PANELMILISECONDSTOTALTAILOFFSET );

/*  QUERRY POPUP Where is the Local Working Directory with Balcon & FFMPEG? */
  QUERRYWORKINGPATH = new Dialog;
  QUERRYWORKINGPATH.title = "Set Local Working Directory with Balcon & FFMPEG?";
  var WORKINGPATH = new Label;
  WORKINGPATH.text = "Where is the Local Working Directory with Balcon & FFMPEG?";
/*  WORKINGPATH.minimum = 0; */
/*  WORKINGPATH.maximum =  2000 ; */
  WORKINGPATH.value = project.getPanelDuration ( CURRENTPANEL[0] );
 QUERRYWORKINGPATH.add( WORKINGPATH );

/*  QUERRY POPUP */



/*  QUERRY POPUP */
/*  QUERRY POPUP */

/*  QUERRY POPUP Frames per Panel*/
  QUERRYFRAMES = new Dialog;
  QUERRYFRAMES.title = "Set New Panel Duration";
  var FRAMELENGTH = new NumberEdit;
  FRAMELENGTH.text = "Panel Length";
  FRAMELENGTH.minimum = 0;
  FRAMELENGTH.maximum =  2000 ;
  FRAMELENGTH.value = project.getPanelDuration ( CURRENTPANEL[0] );
 QUERRYFRAMES.add( FRAMELENGTH );


/*   Begin Steps*/
    if ( QUERRYWORKINGPATH.exec() )
  {
    } 
    if ( QUERRYPANELMILISECONDSTOTALTAILOFFSET.exec() )
  {
    } 

  if ( QUERRYFRAMES.exec() )
  {
    var SETFRAME = FRAMELENGTH.value;

    scene.beginUndoRedoAccum("Change Panel Duration");

    for ( var i = 0; i < CURRENTPANEL.length; ++ i )
    {
        project.setPanelDuration( CURRENTPANEL[i], SETFRAME);
    } 

    scene.endUndoRedoAccum();
  }
}
